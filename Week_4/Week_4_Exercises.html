The Gibbs sampler

Example

# Function to sample from conditional distributions 
sample_x2 = function(k=10000){
  # Specify parameters:
  n     = 16
  alpha = 2
  beta  = 10
  
  # Initialise:
  x = NULL
  y = NULL
  
  y[1] = 1
  x[1] = 0
  
  for (i in 2:k){
    x[i] = rbinom(n=1,size=n,prob=y[i-1])
    y[i] = rbeta(n=1,shape1=(x[i]+alpha),shape2=(n-x[i]+beta))
  }
  
  return(list(x=x,y=y))
}

n = 16
draws = sample_x2(k=100000)
x_full = draws$x
y_full = draws$y

xkeep2   = x_full[floor(100000/2):100000]
ns       = length(xkeep2)
xprobs_2 = sapply(0:n,function(x){sum((xkeep2==x))/ns})
alpha = 2
beta_par = 10
theoretical = sapply(0:n, function(x){
  choose(n,x)*beta(x+alpha,n-x+beta_par)/beta(alpha,beta_par)
})

plot(0:n, 
     theoretical, 
     type = "l",
     lwd  = 2,
     col  = "purple",
     xlab = "x", 
     ylab = expression(pi(x)), 
     main = "Theoretical vs Gibbs.")
lines(0:n, 
      xprobs_2, 
      type = "l",  
      lty  = 2, 
      lwd  = 2,
      col  = "deeppink")
legend("topright", 
       legend = c("Theoretical", "Gibbs"),
       col = c("purple", "deeppink"), 
       lty = c(1,2), 
       bty = "n")

# Function to sample from conditional distributions 
gibbs_sampler_fxn = function(k=5000){
  # Specify parameters:
  p     = 0.5
  
  # Initialise:
  x = NULL
  y = NULL
  
  y[1] = 5
  x[1] = -5
  
  for (i in 2:k){
    x[i] = rnorm(n=1,p*y[i-1],1-p^2)
    y[i] = rnorm(n=1,p*x[i],1-p^2)
  }
  
  return(list(x=x,y=y))
}

p = 0.5
draws = gibbs_sampler_fxn(k=5000)
x_full = draws$x
y_full = draws$y

xkeep    = x_full[floor(5000/2):5000]
nsamp    = length(xkeep)
xprobs   = sapply(0:n,function(x){sum((xkeep==x))/nsamp})
